const fs = require("fs");

// Đọc toàn bộ stdin
const input = fs.readFileSync(0, "utf8").trim();
const n = Number(input);

function factorial(n) {
  if (n < 0) return undefined;
  if (n === 0) return 1;
  let result = 1;
  for (let i = 2; i <= n; i++) {
    result *= i;
  }
  return result;
}

// In ra kết quả duy nhất
console.log(String(factorial(n)));
